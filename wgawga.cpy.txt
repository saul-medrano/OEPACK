PRE'DSP'TOTALS:
    CIF UP'CHARGES = "Y" THEN
        CIF WGA'PRGNAM="OE1WGA" THEN
            IF OEC'SKIP'UP # "Y" THEN
                CALL SHOW'UP'CHARGE        !!Must compute 1st as result is
            ENDIF
        ENDCIF
        II = HDR'LIN'TOTAL + TOTAL'UP'CHARGE
        ORDER'TOTALS = II/100 USING "$$#,###.##-"
    CELSE
        ORDER'TOTALS = HDR'LIN'TOTAL/100 USING "$$#,###.##-"
!!Ben -- the HDR screen includes tax and shipping, etc. -- Invoice total
!!         this screen is supposed to show the total of the NET seen in
!!         the far right column labeled NET ... so that people entering an
!!         can see the order totals as they.  The invoice total is not really
!!         valid until they invoice :-)
!!      IF ORDER'TOTALS # CALC'HTOTAL THEN
!!          \HDRSUM INVOICE="CALC'HTOTAL"
!!          ORDER'TOTALS = CALC'HTOTAL/100 USING "$$#,###.##-"
!!      ENDIF
    ENDCIF
    \STRIP ORDER'TOTALS 3
    IF ARCHIVE'FLAG = "A" THEN
        COPYF "LHSKEY",LHS2
        CAPTTL = 0
        LHS2'KHDR'LOC'NO = HDR'LOC'NO
        LHS2'KDOC'TYPE = HDR'DOC'TYPE
        LHS2'KHDR'NO = HDR'NO
        LHS2'KSHIPMENT = HDR'SHIPMENT
        LHS2'KNO = 0
        USE LHS2 INTO LIN2'REC KEY LHS2'KEY
        DO
            GETNXT LHS2 INTO LIN2'REC
            \CHKKEY LHS ALTPRE="LIN2" ALTKEY="LHS2"
            IF LHS2'CTL # 0 EXIT
            CAPTTL = CAPTTL + LIN2'USR'OQTY
        LOOP
    ENDIF
!!  IF ARCHIVE'FLAG = "C" THEN
        ORDER'TOTALS = STR(CAPTTL) + "/" + ORDER'TOTALS
!!  ENDIF                     !!part of total display.
    CALL DSP'TOTALS
RETURN

PRE'DSP'DOZENS:
    FOR II = 1 TO 6
        CALL DSP'DOZENSII
    NEXT II
RETURN

DSP'DOZENSII:
    CIF USE'CAP = "Y" THEN
        IF (INC'TS'SUBI = 4 AND II > 1) THEN
            RETURN
        ENDIF
    ENDCIF
    CASE
    IF WGA'AFR(II) = "Y" THEN
        DZPC(II)="   *  "
    ELSE
    IF WGA'TITLES(II) = "" THEN
        DZPC(II) = SPACE(6)
    ELSE
        IF INC'DZ'PIECES # "N" THEN
            DZPC(II) = (ABS(WGA'DOZENS(II)) USING "###") + "/" + ABS(WGA'PIECES(II)) + SPACE(2)
        ELSE
            DZPC(II) = WGA'PIECES(II) + SPACE(6)
        ENDIF
    ENDCASE
    CIF WGA'PRGNAM="OE1HDR" OR WGA'PRGNAM="OE1WGA" THEN     !![1]
          ON II CALL DSP'DOZENS1,DSP'DOZENS2,DSP'DOZENS3,       &
                                DSP'DOZENS4,DSP'DOZENS5,DSP'DOZENS6
    ENDCIF
RETURN

MAP1 AUTO'TRK'NO,B,3

HANDLE'ODD'STUFF:
    IF ECFLD = lastfn AND AUTO'DXL = 0 THEN
        IF INX2 = FN'12 THEN
            XCALL GETENV,"PRHURL",ENTRY
            IF ENTRY = "" THEN
                INX2 = FN'C12
            ELSE
                RETURN
            ENDIF
        ENDIF
        IF INX2 = FN'5 THEN INX2 = FN'C5
        IF INX2 = FN'C5 OR INX2 = FN'C12 THEN
            IF PRH'ODD'TRK'NO # 0 THEN
                AUTO'DXL = 1
                INX2 = FN'1
            ELSE
                ENTRY = "No odd size available!"
                GOTO DO'ERR
            ENDIF
        ELSE
        IF INX2 = FN'S5 OR INX2 = FN'S12 THEN
            AUTO'DXL = 99
            INX2 = FN'1
        ENDIF
        ENDIF
        IF AUTO'DXL # 0 THEN
            AUTO'TRK'NO = 0
            IF PRH'ODD'TRK'NO # 0 THEN
                IF AUTO'DXL = 99 THEN
                    II = PRH'TRK'NO
                    PRH2'REC = PRH'REC
                    DO WHILE PRH2'ODD'TRK'NO # 0 AND PRH2'ODD'TRK'NO # II
                        PRH2'KTRK'NO = PRH2'ODD'TRK'NO
                        USE PRH2 INTO PRH2'REC KEY PRH2'KEY
                    LOOP
                    IF PRH2'ODD'TRK'NO = II THEN
                        AUTO'TRK'NO = PRH2'TRK'NO
                    ENDIF
                ELSE
                    AUTO'TRK'NO = PRH'ODD'TRK'NO
                ENDIF
                IF AUTO'TRK'NO = 0 THEN
                    AUTO'DXL = 0
                    INX1 = I'RT : INX2 = 0 : IER1 = 0
                ENDIF
            ENDIF
        ENDIF
    ENDIF
RETURN

CHG'ODD'TRK'NO:
    IF AUTO'TRK'NO # 0 THEN
        LIN'TRK'NO = AUTO'TRK'NO
    ENDIF
    AUTO'DXL = 0
    ECFLD = 2
    INX1 = I'RT : INX2 = 0 : IER1 = 0
RETURN

E'DOZENS:
    arrowc = |EX'EDD| OR arrowc
    IF EDSPSW # 0 GOTO EDIT'DOZENS
    CIF WGA'PRGNAM="OE1WGA" THEN
        IF PG = |NM| AND SELECT(|NM|) = 1 AND IER1 # 0 THEN
            CALL HANDLE'ODD'STUFF
            IF IER1 = 0 RETURN
        ENDIF
    ENDCIF
    IF PG = 3 THEN
        VV = INT((ECFLD+1)/2)
    ELSE
        \GETFLD |SCREEN| WGA'DOZENS(1) V''TMP
        VV = INT((ECFLD+2-|V''TMP|)/2)
    ENDIF
    IF INX1 = I'TAB THEN
        WGA'DOZENS(VV) = DOZENS(VV) MIN 864
        IF DOZENS(VV) = 0 THEN
            WGA'PIECES(VV) = PIECES(VV) MIN 11
        ELSE
            WGA'PIECES(VV) = 0
        ENDIF
        CALL PRE'DSP'DOZENS
        IER1 = 0
    ENDIF
    \EAMT TYPE="HDR'DOC'TYPE" FLDNAM=WGA'DOZENS(VV) &
                                DOERR="GOTO" APID="AR" DO'TSTFUNC="N"
    IF (INX1 = I'UA OR INX1 = I'HME) AND EFCHG # 0 THEN
        CALL UPDATE'DOZENS'PIECES
    ENDIF
    CALL EDIT'DOZENS
    CIF WGA'PRGNAM="OE1WGA" THEN
        IF (lastchb = 20 OR lastchb = 18 OR INX2 = FN'11 OR INX1 = I'HME OR &
                            lastchb = 5) AND PG = 2 AND EDSPSW = 0 THEN
            GOTO E'BOTTOMS
        ENDIF
    ENDCIF
RETURN

E'PIECES:

    IF EDSPSW # 0 GOTO EDIT'PIECES
    CIF WGA'PRGNAM="OE1WGA" THEN
        IF PG = |NM| AND SELECT(|NM|) = 1 AND IER1 # 0 THEN
            CALL HANDLE'ODD'STUFF
            IF IER1 = 0 RETURN
        ENDIF
    ENDCIF
    CALL TST'FUNC
    IF TST'FUNC = 1 THEN
        II = 0
        FOR I = 1 TO 6
            II = II + WGA'PIECES(I) + (12*WGA'DOZENS(I))
        NEXT I
        CASE
        IF SAVE'ORG'INV'AMT # 0 THEN
            STRING'AMT = II/100 USING "##,###,###.##-"
            \STRIP STRING'AMT 3
            IF HDR'DOC'TYPE = 2 AND ABS(II) > ABS(SAVE'ORG'INV'AMT) THEN
                ENTRY = "Credit cannot exceed Invoice amt of "+STRING'AMT
                CALL DO'ERR
                IER1 = 1
            ENDIF
        ELSE
        IF SAVE'ATQ'AMT # 0 THEN
            STRING'AMT = II/100 USING "##,###,###.##-"
            \STRIP STRING'AMT 3
            IF HDR'DOC'TYPE = 2 AND ABS(II) > ABS(SAVE'ATQ'AMT) THEN
                ENTRY = "Refund cannot exceed Credit Card auth amt of "+STRING'AMT
                CALL DO'ERR
                IER1 = 1
            ENDIF
        ENDCASE
        IF II # LIN'ORDER'QTY THEN
            WGA'SPACE = SPACE               !!This is mandatory!!!
        ENDIF
        IF SELECT(|NM|) = 1 OR SELECT(|NM|) = 2 OR XXX'USR'OQTY # 0 THEN
            CALL EDIT'PIECES
            ECFLD = 99              !!This'll speed it up
        ELSE
            CALL TST'FUNC'KLUDGE
        ENDIF
        RETURN
    ENDIF
    IF ENTRY[-1,-1] = "#" THEN              !![5] this if
        II = VAL(ENTRY[1,-2])
        II = II * LIN'UOM'FACTOR(1)
        ENTRY = VAL(II)
        WGA'PIECES(VV) = II
    ENDIF
    IF INC'DZ'PIECES # "N" THEN
        IF PG = 3 THEN
            VV = INT(ECFLD/2)
        ELSE
            \GETFLD |SCREEN| WGA'DOZENS(1) V''TMP
            VV = INT((ECFLD+1-|V''TMP|)/2)
        ENDIF
        IF INX1 = I'TAB THEN
            IF DOZENS(VV) = 0 THEN
                WGA'PIECES(VV) = PIECES(VV) MIN 11
            ELSE
                PIECES(VV) = 11
            ENDIF
            CALL PRE'DSP'DOZENS
            IER1 = 0
        ELSE
            IF WGA'PIECES(VV) >= 12 THEN
                IF ABS(WGA'DOZENS(VV) + FIX(WGA'PIECES(VV)/12)) > 999 THEN
                    ENTRY = "You cannot exceed 999 dozen 11 pieces"
                    CALL DO'ERR
                    WGA'PIECES(VV) = 0
                    RETURN
                ENDIF
                WGA'DOZENS(VV) = WGA'DOZENS(VV) + FIX(WGA'PIECES(VV)/12)
                WGA'PIECES(VV) = WGA'PIECES(VV) - FIX(WGA'PIECES(VV)/12) * 12
                EMAX = 6                !![4] (had dsp here)
            ENDIF
        ENDIF
        IF EMAX # 2 THEN                        !![4]
            II = VV
            CALL DSP'DOZENSII
        ENDIF
    ELSE
        IF PG = 3 THEN
            VV = ECFLD
        ELSE
            \GETFLD |ASCREEN| WGA'PIECES(1) V''TMP
            VV = ECFLD-|V''TMP|+1
        ENDIF
        IF INSTR(1,ENTRY,".") # 0 THEN
           WGA'PIECES(VV) = INT(ENTRY) * 12 + ENTRY[INSTR(1,ENTRY,".")+1,-1]
        ENDIF
        IF INSTR(1,ENTRY,"/") # 0 THEN
           WGA'PIECES(VV) = INT(ENTRY) * 12 + ENTRY[INSTR(1,ENTRY,"/")+1,-1]
        ENDIF
    ENDIF
    IF ZZZW'DOZENS(VV) # WGA'DOZENS(VV) THEN EFCHG = 1
    IF ZZZW'PIECES(VV) # WGA'PIECES(VV) THEN EFCHG = 1
    IF EFCHG # 0 CALL UPDATE'DOZENS'PIECES
    IF AUTO'DXL # 0 AND XXX'ORDER'QTY = 0 GOTO CHG'ODD'TRK'NO
    CIF WGA'PRGNAM="OE1WGA" THEN
        IF (lastchb = 20 OR lastchb = 18 OR INX2 = FN'11 OR INX1 = I'HME OR &
                            lastchb = 5) AND PG = 2 AND EDSPSW = 0 THEN
            GOTO E'BOTTOMS
        ENDIF
    ENDCIF

    CIF USE'CAP = "Y" THEN
        SYMBOL CAPKLUDGE = "INC'TS'SUBI"
    CELSE
        SYMBOL CAPKLUDGE = "5"
    ENDCIF
    IF LIN'UOM'FACTOR(1) = 0 THEN
        ENTRY = "Line invalid:  UOM'FACTOR=0"
        GOTO DO'ERR
    ENDIF
    IF (VV = LAST'GAFR'II OR XXX'USR'OQTY # 0) AND &
((((INT(XXX'USR'OQTY/LIN'UOM'FACTOR(1))*LIN'UOM'FACTOR(1))#XXX'USR'OQTY) AND &
                            |CAPKLUDGE| # 4) OR XXX'USR'OQTY = 0) AND &
                            INX2 # FN'7 AND INX2 # FN'12 AND &
                            (VV # FIRST'GAFR'II OR INX1 # I'UA) AND &
                            INX2 # FN'Y THEN
        IF INX1 # I'UA THEN
TST'FUNC'KLUDGE:
            IF XXX'USR'OQTY = 0 THEN
                ENTRY = "MUST enter an order quantity"
            ELSE
                ENTRY = "Order quantity must be a factor of "+STR(LIN'UOM'FACTOR(1))
            ENDIF
            IF TST'FUNC = 1 THEN
                ENTRY = ENTRY + " Ctrl-Z to DELETE!"
            ENDIF
            CALL DO'ERR
        ELSE
            IER1 = 0
        ENDIF
        IF INC'DZ'PIECES # "N" THEN
            \GOTO |SCREEN| WGA'DOZENS(1)
        ELSE
            \GOTO |ASCREEN| WGA'PIECES(1)
        ENDIF
        IF INX1 = 3 AND INX2 = 0 THEN
            ECFLD = ECFLD - 1
        ELSE
            ECFLD = ECFLD + 1
            INX1 = I'RT
            INX2 = 0
        ENDIF
        RETURN                                  !!KLUDGE
    ENDIF

    CASE
    IF SAVE'ATQ'AMT # 0 AND ABS(HDR'LIN'TOTAL) > ABS(SAVE'ATQ'AMT) THEN
        STRING'AMT = SAVE'ATQ'AMT/100 USING "##,###,###.##-"
        \STRIP STRING'AMT 3
        ENTRY = "*** Refund exceeds CC authorization of "+STRING'AMT
        CALL DO'ERR
!!      IER1 = 1        !!will let it continue to next field after error
    ELSE
    IF SAVE'ORG'INV'AMT # 0 AND ABS(HDR'LIN'TOTAL) > ABS(SAVE'ORG'INV'AMT) THEN
        STRING'AMT = SAVE'ORG'INV'AMT/100 USING "##,###,###.##-"
        \STRIP STRING'AMT 3
        ENTRY = "Credit cannot exceed invoice amount of "+STRING'AMT
        CALL DO'ERR
!!      IER1 = 1        !!will let it continue to next field after error
    ENDCASE

    GOTO EDIT'PIECES
RETURN

UPDATE'DOZENS'PIECES:
        EFCHG = 0
        IF ZZZW'DOZENS(VV) # WGA'DOZENS(VV) THEN EFCHG = 1
        IF ZZZW'PIECES(VV) # WGA'PIECES(VV) THEN EFCHG = 1
        PG = 3
        SUB'KNO = VV
        CIF WGA'PRGNAM="OE1WGA" THEN
            CALL SEE'IF'HPRICE
            O'HPRICE = SF
        ENDCIF
        IF ZZZW'PIECES(VV) # 0 OR ZZZW'DOZENS(VV) # 0 THEN
            TMP'RCN(3) = VV
            CALL PRE'READ3
            IF WGA'PIECES(VV) # 0 OR WGA'DOZENS(VV) # 0 THEN
                CALL WGA'EAMT
                IF INC'DZ'PIECES # "N" THEN
                    XXX'USR'OQTY = (WGA'DOZENS(VV)*12) + &
                                                        WGA'PIECES(VV)
                ELSE
                    XXX'USR'OQTY = WGA'PIECES(VV)
                ENDIF
                XXX'ORDER'QTY = XXX'USR'OQTY
                PG = |NM|
                CALL SET'COM'QTY
                PG = 3
                CALL SET'COM'BO
                IF XXX'USR'OQTY < 0 AND LIN'RPROC = 0 THEN
                    IF HDR'DOC'TYPE # OEC'INT'CR THEN
                        CALL RPROC'PUSH
                        LIN'RPROC = XXX'RPROC
                    ELSE
                        LIN'RPROC = 2
                        XXX'RPROC = 2
                    ENDIF
                ENDIF
                CALL PRE'WRITE3
            ELSE
                CALL DEL'RECORD3
                SUB'SPACE = SPACE
            ENDIF
        ELSE
!PIECES'KLUDGE:
            IF WGA'PIECES(VV) # 0 OR WGA'DOZENS(VV) # 0 THEN
                CALL WGA'EAMT
                CALL DO'DUP3
                CALL SAVTOTAL3
                ZZZ3'SPACE = SPACE              !!TRICK UPDATE'QTY'REC
                ZZZX'SPACE = SPACE              !!DITTO
                        
                SUB'NO = VV
                SUB'ATT'NO1 = VV
                IF INC'DZ'PIECES # "N" THEN
                    XXX'USR'OQTY = (WGA'DOZENS(VV)*12) + &
                                                        WGA'PIECES(VV)
                ELSE
                    XXX'USR'OQTY = WGA'PIECES(VV)
                ENDIF
                XXX'ORDER'QTY = XXX'USR'OQTY
                PG = |NM|
                CALL SET'COM'QTY
                PG = 3
                CALL SET'COM'BO

                                ENDIF

                            ENDIF

                AVAILABLE = DOZENS(II) * 12 + PIECES(II)
